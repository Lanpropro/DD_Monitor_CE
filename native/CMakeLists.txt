cmake_minimum_required(VERSION 3.24)
project(ddm_native_danmaku LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PYBIND11_FINDPYTHON ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
include(CTest)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
endif()
add_library(danmaku_engine STATIC src/engine.cpp)
target_include_directories(danmaku_engine PUBLIC src)
pybind11_add_module(_native_danmaku src/bindings.cpp src/host.cpp src/renderer.cpp)
target_link_libraries(_native_danmaku PRIVATE danmaku_engine d3d11 dxgi dcomp d3dcompiler)
target_compile_definitions(_native_danmaku PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
install(TARGETS _native_danmaku LIBRARY DESTINATION ddm RUNTIME DESTINATION ddm)
if(BUILD_TESTING)
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE danmaku_engine)
  add_test(NAME engine COMMAND engine_tests)
endif()
if(BUILD_TESTING AND WIN32)
  add_executable(renderer_tests tests/renderer_tests.cpp src/host.cpp src/renderer.cpp)
  target_link_libraries(renderer_tests PRIVATE danmaku_engine d3d11 dxgi dcomp d3dcompiler)
  target_compile_definitions(renderer_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_test(NAME renderer COMMAND renderer_tests)
endif()
